form.addEventListener("submit", async (event) => { event.preventDefault(); const firstName = document.getElementById("first-name").value.trim(); const lastName = document.getElementById("last-name").value.trim(); const phone = document.getElementById("phone").value.trim(); const email = document.getElementById("email").value.trim(); const password = document.getElementById("password").value; if (!firstName || !lastName || !email || !password) { showMessage("Please complete all required fields."); return; } const client = window.oakmontSupabase; if (!client || !client.auth) { showMessage( "Supabase client missing. Check supabase-client.js and its script loading order." ); console.error("Oakmont diagnostic: window.oakmontSupabase is unavailable."); return; } submitButton.disabled = true; submitButton.textContent = "Creating Account…"; showMessage("Connecting to the registration service…"); let timeoutId; try { const registrationRequest = client.auth.signUp({ email, password, options: { data: { first_name: firstName, last_name: lastName, phone: phone || null }, emailRedirectTo: "https://oakmontgroup.github.io/oakmont-private-banking/dashboard.html" } }); const timeoutRequest = new Promise((_, reject) => { timeoutId = setTimeout(() => { reject( new Error( "TIMEOUT: Supabase did not respond within 15 seconds. Check your internet connection, project URL, publishable key, and browser console." ) ); }, 15000); }); const { data, error } = await Promise.race([ registrationRequest, timeoutRequest ]); if (error) { console.error("Supabase signup error:", error); showMessage("Registration failed: " + error.message); return; } if (data?.user && data.session) { showMessage("Account created. Redirecting to your dashboard…", "success"); window.location.replace("dashboard.html"); return; } if (data?.user) { showMessage( "Your account was created. Check your email to confirm it, then sign in.", "success" ); return; } showMessage( "Supabase responded, but no user was returned. Check the browser console." ); console.error("Unexpected signup response:", data); } catch (error) { console.error("Oakmont registration diagnostic:", error); showMessage( error?.message || "Registration failed unexpectedly. Open the browser console for details." ); } finally { clearTimeout(timeoutId); submitButton.disabled = false; submitButton.textContent = "Create Account"; } });
